\usepackage{fontspec}
\usepackage{xeCJK}

% 日本語フォント設定
\setCJKmainfont{IPAGothic}

% 絵文字フォント設定（フォールバック）
\newfontfamily\emojifont{Noto Color Emoji}[Renderer=Harfbuzz]

% フォントマッピング: 絵文字範囲をNoto Color Emojiにマッピング
\XeTeXinterchartokenstate=1
\newXeTeXintercharclass\emojiclass

% 絵文字の主要なUnicode範囲を指定
% U+2600-U+26FF: Miscellaneous Symbols
\count255=`\^^2600
\loop
  \XeTeXcharclass\count255=\emojiclass
  \advance\count255 by 1
  \ifnum\count255<`\^^2700
\repeat
